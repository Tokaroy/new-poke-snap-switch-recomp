# fixture: post_tutorial
# Save-state check on Florio (docs/save-states.md): ride, save, take photos, load, and see the film counter and the ride come back.
# Run: powershell -File host\run_boot.ps1 -Route savestate_florio -Timeout 420
# Pass: boot_stderr.log has "state save 'ck': ok" (film counter 70 then: the boot taps took two photos), then for each of the 3 rounds photos until the counter
# is at most 67, a "state load 'ck': ok", and "wait '!film<=69' met" (the HUD reports 70 again: the load rewound the game and the HLE film mirror was reset),
# and finally "end of route reached" (exit code 0).
# A save=/load= token fires once when its line starts. The entries before a `wait` run while it waits, so the request line comes first, then
# `wait state=saves>=n` / `wait state=loads>=n` (counters of finished requests; they cannot miss a request that ended between two polls).
repeat @f300 @f30000 120 8 A
wait ui=map_select timeout=240
repeat @f60 @f30000 120 8 A
wait ride>=3 timeout=240
@f10 8 save=ck
wait state=saves>=1 timeout=60
# round 1: shutter taps until the counter drops, then load and expect the counter back at 72
repeat @f0 @f30000 24 4 A
wait film<=67 timeout=60
@f10 8 load=ck
wait state=loads>=1 timeout=60
wait !film<=69 timeout=10
# round 2
repeat @f0 @f30000 24 4 A
wait film<=67 timeout=60
@f10 8 load=ck
wait state=loads>=2 timeout=60
wait !film<=69 timeout=10
# round 3
repeat @f0 @f30000 24 4 A
wait film<=67 timeout=60
@f10 8 load=ck
wait state=loads>=3 timeout=60
wait !film<=69 timeout=10
wait ride timeout=30
end @f120
